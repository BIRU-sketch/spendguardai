# 1. Point standard OpenAI client to SpendGuard Gateway
import openai

client = openai.OpenAI(
    base_url="http://localhost:8000/v1",  # SpendGuard Gateway
    api_key="sk-live-spendguard-production-key",
)

# 2. Attach run_id and agent_id headers for hard cost control
response = client.chat.completions.create(
    model="gpt-4o",  # or claude-3-5-sonnet, gemini-2.0-flash
    messages=[{"role": "user", "content": "Execute autonomous agent task."}],
    extra_headers={
        "X-SpendGuard-Run-Id": "run_crawler_994",
        "X-SpendGuard-Agent-Id": "web_researcher_v1",
        "X-SpendGuard-Run-Budget-Cents": "50",  # $0.50 hard cap
        "X-SpendGuard-Max-Steps": "25",         # Runaway limit
    },
)

# SpendGuard handles pre-flight locks, true-up refunds,
# and stops runaway loops before you get billed.
