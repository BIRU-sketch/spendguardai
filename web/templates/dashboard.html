SDK Quickstart

Point OpenAI client libraries directly to your SpendGuard endpoint.

from openai import OpenAI

client = OpenAI(
    base_url="{{ gateway_url }}",
    api_key ="YOUR_SPENDGUARD_API_KEY",
)

response = client.chat.completions.create(
    model="gpt-4o-mini",
    messages=[{"role": "user", "content": "Hello SpendGuard!"}],
    extra_headers={"X-End-User-Id": "customer_123"},
)
print(response.choices[0].message.content)
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "{{ gateway_url }}",
  apiKey:  "YOUR_SPENDGUARD_API_KEY",
});

const response = await client.chat.completions.create({
  model: "gpt-4o-mini",
  messages: [{ role: "user", content: "Hello SpendGuard!" }],
}, {
  headers: { "X-End-User-Id": "customer_123" },
});
console.log(response.choices[0].message.content);
curl -X POST "{{ gateway_url }}/chat/completions" \
  -H "Authorization: Bearer YOUR_SPENDGUARD_API_KEY" \
  -H "Content-Type: application/json" \
  -H "X-End-User-Id: customer_123" \
  -d '{
    "model": "gpt-4o-mini",
    "messages": [{"role": "user", "content": "Hello SpendGuard!"}]
  }'
